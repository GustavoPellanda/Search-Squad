cmake_minimum_required(VERSION 3.8)
project(explorer)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(squad_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)

find_package(rclcpp_action REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)

add_executable(reactive_controller src/reactive_controller.cpp)
ament_target_dependencies(reactive_controller squad_interfaces rclcpp geometry_msgs sensor_msgs)
add_library(room_geometry src/room_geometry.cpp)
target_include_directories(room_geometry PUBLIC include)
ament_target_dependencies(room_geometry nav_msgs geometry_msgs)
add_executable(room_explorer src/room_explorer.cpp)
target_include_directories(room_explorer PRIVATE include)
target_link_libraries(room_explorer room_geometry)
ament_target_dependencies(room_explorer std_msgs rclcpp rclcpp_action squad_interfaces sensor_msgs
  nav_msgs nav2_msgs visualization_msgs tf2 tf2_ros)
install(TARGETS reactive_controller room_explorer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_room_geometry test/test_room_geometry.cpp)
  target_include_directories(test_room_geometry PRIVATE include)
  target_link_libraries(test_room_geometry room_geometry)
  ament_target_dependencies(test_room_geometry nav_msgs geometry_msgs)
endif()
ament_package()
