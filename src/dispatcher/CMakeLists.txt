cmake_minimum_required(VERSION 3.8)
project(dispatcher)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(squad_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(entrance_dispatcher src/entrance_dispatcher.cpp)
ament_target_dependencies(entrance_dispatcher
  rclcpp rclcpp_action geometry_msgs nav2_msgs nav_msgs lifecycle_msgs squad_interfaces tf2 tf2_ros
)
install(TARGETS entrance_dispatcher DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
