cmake_minimum_required(VERSION 3.8)
project(mapping)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(sim_pose_adapter src/sim_pose_adapter.cpp)
ament_target_dependencies(sim_pose_adapter rclcpp geometry_msgs nav_msgs sensor_msgs gazebo_msgs tf2 tf2_ros tf2_geometry_msgs)
add_executable(occupancy_mapper src/occupancy_mapper.cpp)
ament_target_dependencies(occupancy_mapper rclcpp geometry_msgs nav_msgs sensor_msgs gazebo_msgs tf2 tf2_ros tf2_geometry_msgs)

install(TARGETS sim_pose_adapter occupancy_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME} OPTIONAL)
ament_package()
